## 3. Execution

**Fail-closed rule.** Every check runs in `gate.sh` (§8), which the executor extracts from this contract into `/tmp/e2e-003` (outside the repository; never into `/tmp/e2e-002`, which holds E2E-002's evidence) and runs with Git Bash from the repository's top directory. A gate mode either ends with `GATE PASSED: …` and exit status 0, or stops with a `GATE FAILED: …` line and a non-zero exit status. Only `ci` can also end with `GATE WAITING: …` and exit status 3: the run has not completed or GitHub could not be read; that is not a result, and the executor runs the same command again later. Every passing mode appends its line to `/tmp/e2e-003/ledger`, and the gate accepts in the Task record only gate lines it printed itself. The branch, both commits, both pushes and the pull request are made only in the same command line as the gate mode that guards them, joined by `&&`. Any `GATE FAILED`, any non-zero exit status other than the `GATE WAITING` of `ci`, or any output other than the expected one stops the Task (T8): the executor records it, reports it and waits. It never re-runs a check to obtain a pass, never edits or bypasses the gate, and never runs `git switch -c`, `git commit`, `git push` or `gh pr create` outside the guarded command lines of Steps 1, 5, 6 and 7, and never `gh run rerun` or `gh workflow run` at all.

  [ "$msg" = "$subject" ] || fail "commit $c: the message is not exactly: $subject"
}
# The two paths of 2.1 against the base: in the index (stage "index") or in HEAD.
paths_ok() {
  local v
  if [ "$1" = index ]; then v=$(git diff --cached --no-renames --name-status "$BASE"); else v=$(git diff --no-renames --name-status "$BASE" HEAD); fi
  [ "$(printf '%s\n' "$v" | awk '{print $1, $2}' | LC_ALL=C sort)" = "$(printf '%s\n' "$PATHS" | LC_ALL=C sort)" ] || fail "the change set against the base is not exactly the two paths of 2.1"
}
# The workflow and the Task record as they are staged (stage "index") or committed (HEAD).
content_ok() { # where (index or HEAD), stage (content or evidence), then for evidence: head, run, pull request
  local where=$1 stage=$2 src
  if [ "$where" = index ]; then src=":"; else src="HEAD:"; fi
  git show "$BASE:$WORKFLOW" > "$W/ci.base" && git show "$src$WORKFLOW" > "$W/ci.staged" || fail "ci.yml cannot be read"
  check pin same "$(winpath "$W/ci.base")" "$(winpath "$W/ci.staged")" || fail "ci.yml (Step 3)"
  git show "$src$TASK" > "$W/record.staged" 2> /dev/null || fail "the Task record is not in the $where"
  check record "$(winpath "$(contract)")" "$(winpath "$W/record.staged")" "$stage" "$(cat "$W/contract.sha256")" "$(winpath "$W/ledger")" "${@:3}" || fail "the Task record ($stage)"
}

init() {
  [ "$#" -eq 2 ] || fail "usage: init <CONTRACT_FILE> <HASH>"
  [ -f "$1" ] || fail "T5: no contract file $1"
  tr -d '\r' < "$1" > "$W/contract.lf.new"
  [ "$(sha256sum < "$W/contract.lf.new" | cut -d' ' -f1)" = "$2" ] || fail "T5: the contract's LF-normalized SHA-256 is not $2"
  mv "$W/contract.lf.new" "$W/contract.lf"
  printf '%s\n' "$2" > "$W/contract.sha256"
  sha256sum < "$0" | cut -d' ' -f1 > "$W/gate.sha256"
  passed "init (contract $2, base $BASE)"
}

pre() {
  in_repo
  identity
  local v
  status_now
  [ -z "$STATUS" ] || fail "T1: the working tree is not clean"
  [ "$(git rev-parse origin/main)" = "$BASE" ] || fail "T3: origin/main is not the base $BASE"
  v=$(git ls-remote origin refs/heads/main) || fail "T3: origin cannot be read"
  [ "${v%%[[:space:]]*}" = "$BASE" ] || fail "T3: main on origin is ${v%%[[:space:]]*}, not the base $BASE"
  ! git rev-parse --verify --quiet "refs/heads/$BRANCH" > /dev/null || fail "T3: the branch $BRANCH exists locally"
  [ -z "$(git ls-remote origin "refs/heads/$BRANCH")" ] || fail "T3: the branch $BRANCH exists on origin"
  git show "$BASE:$WORKFLOW" > "$W/ci.base" || fail "ci.yml cannot be read"
  [ "$(git rev-parse "$BASE:$WORKFLOW")" = 769588b5b51f9ea75737e44c3a5b35a5e8dd8ba5 ] || fail "T4: ci.yml of the base is not the workflow this contract was written for"
  passed "pre (base $BASE)"
}

pin() {
  in_repo
  on_branch
  [ "$(git rev-parse HEAD)" = "$BASE" ] || fail "HEAD is not the base"
  status_now
  [ "$STATUS" = "?? $TASK" ] || fail "only the Task record may be new before Step 3, nothing else changed: $STATUS"
  git show "$BASE:$WORKFLOW" > "$W/ci.base" || fail "ci.yml cannot be read"
  check pin write "$(winpath "$W/ci.base")" "$(winpath "$WORKFLOW")" || fail "T4: ci.yml (Step 3)"
  passed "pin (ci.yml $(git hash-object "$WORKFLOW"): four runs-on ubuntu-24.04 and the comment above jobs:)"
}

content() {
  in_repo
  on_branch
  identity
  local line
  [ "$(git rev-parse HEAD)" = "$BASE" ] || fail "HEAD is not the base: no commit before Step 5"
  status_now
  while IFS= read -r line; do
    case "$line" in "A  $TASK" | "M  $WORKFLOW") ;; *) fail "unstaged, untracked or other change: $line" ;; esac
  done <<< "$STATUS"
  paths_ok index
  git diff --cached --check "$BASE" > "$W/whitespace.out" 2>&1 || fail "whitespace errors: $(head -n 5 "$W/whitespace.out")"
  content_ok index content
  passed "content (ci.yml $(git rev-parse ":$WORKFLOW"); Task record VERIFYING)"
}

committed() {
  in_repo
  on_branch
  local subject=${1:-$SUBJECT_CONTENT} c
  status_now
  [ -z "$STATUS" ] || fail "the working tree is not clean after the commit"
  if [ "$subject" = "$SUBJECT_CONTENT" ]; then c=HEAD; else c=HEAD~1; fi
  [ "$(git rev-parse "$c^@" | tr '\n' ' ')" = "$BASE " ] || fail "the content commit is not one commit on the base"
  commit_ok "$c" "$SUBJECT_CONTENT"
  [ "$(git rev-list --merges "$BASE..HEAD")" = "" ] || fail "a merge commit is on the branch"
  paths_ok HEAD
  if [ "$subject" = "$SUBJECT_CONTENT" ]; then
    content_ok HEAD content
  else
    commit_ok HEAD "$SUBJECT_EVIDENCE"
    [ "$(git diff --no-renames --name-status HEAD~1 HEAD | awk '{print $1, $2}')" = "M $TASK" ] || fail "the evidence commit changes more than the Task record"
    [ "$(git rev-list --count "$BASE..HEAD")" = 2 ] || fail "the branch has more than the content and the evidence commit"
  fi
  passed "committed $(git rev-parse HEAD) $subject"
}

authorized() {
  in_repo
  on_branch
  [ "$#" -eq 1 ] && [[ "$1" =~ ^(push|pull-request)$ ]] || fail "usage: authorized push | pull-request"
  git show "HEAD:$TASK" > "$W/record.authorization" 2> /dev/null || fail "the Task record is not in HEAD"
  check authorized "$(winpath "$W/record.authorization")" "$1" || fail "not authorized: $1 (the record in HEAD $(git rev-parse HEAD))"
  passed "authorized $1 (as recorded in the record in HEAD $(git rev-parse HEAD); the gate cannot prove that the Owner gave it)"
}

pr_body() {
  in_repo
  check prbody "$(winpath "$(contract)")" "$(winpath "$W/pr-body.md")" || fail "the pull request body could not be written"
  passed "pr-body $W/pr-body.md"
}

pr() {
  in_repo
  on_branch
  [ "$#" -eq 1 ] && [[ "$1" =~ ^[0-9]+$ ]] || fail "usage: pr <NUMBER>"
  local head count meta
  head=$(git rev-parse HEAD)
  count=$(git rev-list --count "$BASE..HEAD") || fail "the commits since the base cannot be counted"
  meta=$(gh api "repos/$REPO/pulls/$1" --jq '[.state, (.draft | tostring), .base.ref, .head.ref, .head.sha, (.changed_files | tostring), (.commits | tostring), .title] | join("|")') || fail "pull request #$1 cannot be read"
  [ "$meta" = "open|false|main|$BRANCH|$head|2|$count|$SUBJECT_CONTENT" ] || fail "pull request #$1 is '$meta', not 'open|false|main|$BRANCH|$head|2|$count|$SUBJECT_CONTENT'"
  gh api "repos/$REPO/pulls/$1" --jq .body > "$W/pr-body.actual" || fail "the pull request body cannot be read"
  check prbody "$(winpath "$(contract)")" "" "$(winpath "$W/pr-body.actual")" || fail "the pull request body"
  passed "pr $1 ($count commit(s), 2 files, head $head)"
}

fetch_log() { # job id, file: three tries, 15 seconds apart
  local t
  for t in 1 2 3; do
    if gh api "repos/$REPO/actions/jobs/$1/logs" > "$2" 2> /dev/null && [ -s "$2" ]; then return 0; fi
    [ "$t" -eq 3 ] || sleep 15
  done
  return 1
}

# The full CI of the content commit (attempt 1; this Task starts no re-run): every job on ubuntu-24.04, every browser
# installed, every job success, except a firefox-1280 failure in its test step with the E2E-002 signature when the
# Firefox rule recorded at the content commit (D-R2) is KNOWN-SIGNATURE.
ci() {
  in_repo
  on_branch
  [ "$#" -eq 1 ] && [[ "$1" =~ ^[0-9]+$ ]] || fail "usage: ci <RUN>"
  local run=$1 head info jobs j id file image steps conclusion counts rule note=""
  head=$(git rev-parse HEAD)
  git show "HEAD:$TASK" > "$W/record.rule" 2> /dev/null || fail "the Task record is not in HEAD"
  rule=$(check rule "$(winpath "$W/record.rule")") || fail "the Firefox rule (D-R2)"
  info=$(gh api "repos/$REPO/actions/runs/$run/attempts/1" --jq '[.status, (.conclusion // "none"), .head_sha, .event, .head_branch] | join(" ")') || waiting "GitHub could not be read (run $run attempt 1)."
  case "$info" in "completed "*) ;; *) waiting "run $run attempt 1 is ${info%% *}." ;; esac
  jobs="$W/ci-$run-1.jobs"
  gh api "repos/$REPO/actions/runs/$run/attempts/1/jobs?per_page=100" --jq '.jobs[] | [(.id | tostring), (.conclusion // "none"), .started_at, .name] | join("|")' > "$jobs" || waiting "GitHub could not be read (the jobs of run $run)."
  for j in "${JOBS[@]}"; do
    id=$(awk -F'|' -v n="$j" '$4 == n {print $1}' "$jobs")
    file="$W/ci-$run-1-$(printf '%s' "$j" | tr -c 'A-Za-z0-9.-' '_').log"
    if [ -n "$id" ] && fetch_log "$id" "$file"; then
      image=$(sed -n '/Image: /{s/.*Image: //p;q}' "$file" | tr -d '\r')
      gh api "repos/$REPO/actions/jobs/$id" --jq '.steps[] | [.name, (.conclusion // "none")] | join("|")' > "$file.steps" || waiting "GitHub could not be read (the steps of job $id)."
      echo "log of job $id ($j) saved: $file (runner image: ${image:-not found})"
    else
      : > "$file"
      echo "log of job '$j' not available"
    fi
  done
  check jobs "$(winpath "$jobs")" || fail "run $run: the eight jobs"
  case "$info" in "completed "*" $head push $BRANCH") ;; *) fail "run $run attempt 1 is '$info', not on $head from a push to $BRANCH" ;; esac
  for j in "${JOBS[@]}"; do
    file="$W/ci-$run-1-$(printf '%s' "$j" | tr -c 'A-Za-z0-9.-' '_').log"
    [ -s "$file" ] || waiting "the log of '$j' (run $run) cannot be downloaded yet."
    image=$(sed -n '/Image: /{s/.*Image: //p;q}' "$file" | tr -d '\r')
    [ "$image" = "ubuntu-24.04" ] || fail "the $j job ran on runner image '$image', not ubuntu-24.04"
    conclusion=$(awk -F'|' -v n="$j" '$4 == n {print $2}' "$jobs")
    case "$j" in
      "Browser end-to-end ("*)
        steps=$(cat "$file.steps")
        grep -qxF "Install suite and browser|success" <<< "$steps" || fail "the $j job did not install its browser on ubuntu-24.04 ('Install suite and browser' is not success)"
        counts=$({ sed 's/\x1b\[[0-9;]*m//g' "$file" | grep -oE '[0-9]+ (passed|failed|flaky|skipped|interrupted|did not run)' || true; } | tr '\n' ' ')
        if [ "$conclusion" = success ] && [ "$counts" = "21 passed " ]; then
          echo "$j: ubuntu-24.04, browser installed, 21 passed"
        elif [ "$j" = "$FIREFOX" ] && [ "$rule" = KNOWN-SIGNATURE ] && [ "$conclusion" = failure ] \
          && grep -qxF "Pilot journey, resilience, security, results and exam lifecycle|failure" <<< "$steps"; then
          check signature "$(winpath "$file")" || fail "the $j job failed for a reason other than the E2E-002 signature (test summary '$counts')"
          note="; firefox-1280 failed in its test step with the E2E-002 signature (${counts% }), recorded as a known defect under D-R2"
          echo "$j: ubuntu-24.04, browser installed, test step failed with the E2E-002 signature: $counts"
        else
          fail "the $j job is '$conclusion' with test summary '$counts', not success with '21 passed' (Firefox rule: $rule)"
        fi
        ;;
      *)
        [ "$conclusion" = success ] || fail "the $j job is '$conclusion', not success"
        echo "$j: ubuntu-24.04, success"
        ;;
    esac
  done
  if [ -z "$note" ]; then
    [ "$info" = "completed success $head push $BRANCH" ] || fail "run $run attempt 1 is '$info', not 'completed success $head push $BRANCH'"
  else
    [ "$info" = "completed failure $head push $BRANCH" ] || fail "run $run attempt 1 is '$info', not 'completed failure $head push $BRANCH'"
  fi
  passed "ci $run attempt 1 (head $head; all eight jobs on ubuntu-24.04; five browsers installed$( [ -z "$note" ] && printf '; all eight jobs success' || printf '%s' "$note"))"
}

evidence() {
  in_repo
  on_branch
  identity
  [ "$#" -eq 2 ] && [[ "$1" =~ ^[0-9]+$ ]] && [[ "$2" =~ ^[0-9]+$ ]] || fail "usage: evidence <RUN> <NUMBER>"
  local head
  head=$(git rev-parse HEAD)
  [ "$(git rev-parse "HEAD^@" | tr '\n' ' ')" = "$BASE " ] || fail "HEAD is not the content commit (one commit on the base)"
  commit_ok HEAD "$SUBJECT_CONTENT"
  status_now
  [ "$STATUS" = "M  $TASK" ] || fail "only the Task record may be staged, with nothing unstaged or untracked"
  git show "HEAD:$WORKFLOW" > "$W/ci.head" && git show ":$WORKFLOW" > "$W/ci.index" || fail "ci.yml cannot be read"
  cmp -s "$W/ci.head" "$W/ci.index" || fail "ci.yml is staged with a change"
  content_ok index evidence "$head" "$1" "$2"
  git show "HEAD:$TASK" > "$W/record.content" || fail "the record of the content commit cannot be read"
  check sameauth "$(winpath "$W/record.content")" "$(winpath "$W/record.staged")" || fail "the evidence record changes the approval, authorization or rule"
  passed "evidence (run $1, pull request #$2, head $head)"
}

evidence_committed() {
  in_repo
  on_branch
  [ "$#" -eq 2 ] && [[ "$1" =~ ^[0-9]+$ ]] && [[ "$2" =~ ^[0-9]+$ ]] || fail "usage: evidence-committed <RUN> <NUMBER>"
  committed "$SUBJECT_EVIDENCE" > /dev/null
  content_ok HEAD evidence "$(git rev-parse HEAD~1)" "$1" "$2"
  git show "HEAD~1:$TASK" > "$W/record.content" || fail "the record of the content commit cannot be read"
  check sameauth "$(winpath "$W/record.content")" "$(winpath "$W/record.staged")" || fail "the evidence commit changes the approval, authorization or rule"
  passed "evidence-committed $(git rev-parse HEAD) (run $1, pull request #$2, content commit $(git rev-parse HEAD~1))"
}

mode=${1:-}
[ -n "$mode" ] || fail "usage: gate.sh <mode> [arguments]"
shift
[ "$mode" = init ] || self_check
case "$mode" in
  init) init "$@" ;;
  pre) pre ;;
  pin) pin ;;
  content) content ;;
  committed) committed ;;
  authorized) authorized "$@" ;;
  pr-body) pr_body ;;
  pr) pr "$@" ;;
  ci) ci "$@" ;;
  evidence) evidence "$@" ;;
  evidence-committed) evidence_committed "$@" ;;
  *) fail "unknown mode $mode" ;;
esac
